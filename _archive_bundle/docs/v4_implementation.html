Section Four

Git WorkflowBranching, commits, releases. Designed for solo + AI agents, not for a team of ten.

A small project does not need GitFlow. It needs: main is always deployable, agents work on feature branches, commits are atomic enough that revertible, releases are tagged.

Branch model

  • main — always deployable. Cloudflare Pages auto-deploys from this on every push.
  • feature/agent-a-board etc. — one branch per agent task, named for the brief.
  • release/v0.1.0-stage0 — release branches at milestone tags.
  • No develop branch. No staging branch. Cloudflare Pages preview URLs handle preview deploys per PR automatically.

Commit convention

Conventional Commits format. Agents are instructed to use it.

Section Six

Engine IntegrationStockfish & Maia in Web Workers. The trickiest part of the build, fully spelled out.

Browser chess engines need three things: bundled WASM in /public/engines/, a Web Worker wrapper that speaks UCI, and a React-side bridge that exposes a typed promise-based API. Below is each layer.

Stockfish worker

src/engine/stockfish.worker.ts 
/// <reference lib="webworker" />
// Web Worker that hosts stockfish.wasm and speaks UCI protocol.
// Messages from main thread:  { cmd: 'eval' | 'go' | 'stop', fen?, depth? }
// Messages to main thread:    { type: 'bestmove' | 'info' | 'ready', ... }

let sf: any = null;

async function init() {
  const Stockfish = (await import("stockfish.js")).default;
  sf = await Stockfish();
  sf.addMessageListener(handleEngineLine);
  sf.postMessage("uci");
  sf.postMessage("isready");
}

function handleEngineLine(line: string) {
  if (line === "readyok") {
    self.postMessage({ type: "ready" });
  } else if (line.startsWith("bestmove")) {
    const parts = line.split(" ");
    self.postMessage({ type: "bestmove", move: parts[1] });
  } else if (line.startsWith("info") && line.includes("score")) {
    // Parse "info depth N score cp X pv ..." for eval-only requests
    const cpMatch = line.match(/score cp (-?\d+)/);
    const mateMatch = line.match(/score mate (-?\d+)/);
    self.postMessage({
      type: "info",
      cp: cpMatch ? parseInt(cpMatch[1], 10) : null,
      mate: mateMatch ? parseInt(mateMatch[1], 10) : null,
      raw: line,
    });
  }

src/engine/stockfishBridge.ts 
// Promise-based facade over the Stockfish worker.
// One worker instance reused across the session.

let worker: Worker | null = null;
let ready = false;
const readyWaiters: Array<() => void> = [];

function getWorker(): Worker {
  if (!worker) {
    worker = new Worker(new URL("./stockfish.worker.ts", import.meta.url), { type: "module" });
    worker.onmessage = (e) => {
      if (e.data.type === "ready") {
        ready = true;
        readyWaiters.forEach((r) => r());
        readyWaiters.length = 0;
      }
    };
  }
  return worker;
}

export async function waitForEngine(): Promise<void> {
  if (ready) return;
  getWorker();
  return new Promise((resolve) => readyWaiters.push(resolve));
}

export async function getBestMove(fen: string, depth = 12): Promise<string> {
  await waitForEngine();
  const w = getWorker();
  return new Promise((resolve) => {
    const handler = (e: MessageEvent) => {
      if (e.data.type === "bestmove") {
        w.removeEventListener("message", handler);
        resolve(e.data.move);
      }
    };
    w.addEventListener("message", handler);
    w.postMessage({ cmd: "go", fen, depth });
  });
}

export async function evaluatePosition(fen: string, depth = 14): Promise<{ cp: number | null; mate: number | null }> {
  await waitForEngine();
  const w = getWorker();
  return new Promise((resolve) => {
    let last = { cp: null as number | null, mate: null as number | null };
    const handler = (e: MessageEvent) => {
      if (e.data.type === "info") last = { cp: e.data.cp, mate: e.data.mate };
      if (e.data.type === "bestmove") {
        w.removeEventListener("message", handler);
        resolve(last);
      }
    };
    w.addEventListener("message", handler);
    w.postMessage({ cmd: "eval", fen, depth });
  });
}

Loading order in the PWA

  1. App opens. main.tsx mounts.
  2. Service worker registers and caches the app shell.
  3. Service worker pre-caches /engines/stockfish/* in background.
  4. When player starts calibration, Stockfish worker initializes. First eval takes ~1-2 seconds (one-time load cost).
  5. When player's Elo band is detected post-calibration, the appropriate Maia weight file is fetched in background.
  6. When player faces a character match, Maia worker is initialized with the right weights if not already loaded.

sys.path.append(os.path.dirname(os.path.dirname(__file__)))
from _palette import PANDAVA, NEUTRAL

def clear_scene():
    bpy.ops.object.select_all(action="SELECT")
    bpy.ops.object.delete()

def make_material(name, color, metallic=0.0, roughness=0.5):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs["Base Color"].default_value = color
    bsdf.inputs["Metallic"].default_value = metallic
    bsdf.inputs["Roughness"].default_value = roughness
    return mat

def build_chariot():
    clear_scene()
    
    # Cart body — rounded box
    bpy.ops.mesh.primitive_cube_add(size=1.0, location=(0, 0, 0.5))
    body = bpy.context.active_object
    body.scale = (0.9, 0.5, 0.4)
    bpy.ops.object.modifier_add(type="BEVEL")
    body.modifiers["Bevel"].width = 0.05
    body.modifiers["Bevel"].segments = 3
    
    # Two wheels — spoked
    for y_offset in [0.45, -0.45]:
        bpy.ops.mesh.primitive_torus_add(
            major_radius=0.4, minor_radius=0.04,
            location=(0, y_offset, 0.3),
            rotation=(math.radians(90), 0, 0)
        )
        # Spokes
        for angle in [0, math.pi/3, 2*math.pi/3]:
            bpy.ops.mesh.primitive_cylinder_add(
                radius=0.015, depth=0.8,
                location=(0, y_offset, 0.3),
                rotation=(0, angle, 0)
            )
    
    # Banner pole
    bpy.ops.mesh.primitive_cylinder_add(
        radius=0.04, depth=1.6,
        location=(-0.3, 0, 1.3)
    )
    
    # Banner cloth — slightly curved plane
    bpy.ops.mesh.primitive_plane_add(size=0.7, location=(-0.3, 0, 1.7))
    banner = bpy.context.active_object
    banner.scale = (0.5, 0.01, 0.6)
    # Subdivide and add slight wave
    bpy.ops.object.modifier_add(type="SUBSURF")
    bpy.ops.object.modifier_add(type="WAVE")
    banner.modifiers["Wave"].height = 0.03

def apply_materials():
    wood = make_material("wood", NEUTRAL["wood"], roughness=0.7)
    gold = make_material("gold", PANDAVA["accent"], metallic=0.85, roughness=0.25)
    indigo = make_material("indigo", PANDAVA["primary"], roughness=0.5)
    
    for obj in bpy.data.objects:
        if obj.type != "MESH": continue

        elif "pole" in obj.name.lower() or "cylinder" in obj.name.lower():

Daily rate limiting

Free Gemini tier is 1,500 req/day across the entire app. At Stage 1 (500 DAU × 1 Coach call each), we use 33% of the daily quota. To stay safe:

  1. Hard cap: 5 Coach interactions per player per day. Soft-prompt the player to rest beyond that.
  2. Cache Coach responses by (motif, history_band, position_class) — many players will hit similar Coach moments and the response can be reused with minor variation. ~40% cache hit rate expected at 1K+ DAU.
  3. Pre-generate 200 "canonical" Coach responses for the 10 most common motifs at three player levels. Use them as fallback if Gemini quota exhausted.

import { StyleVector } from "../ml/styleVector";
import { getMaiaPolicy } from "./maiaBridge";
import { getBestMove, evaluatePosition } from "./stockfishBridge";
import { detectMotifInPosition, MOTIF_THRESHOLD } from "./motifClassifier";
import { perturbPolicy } from "../ml/perturbPolicy";

export interface MirrorConfig {
  styleVector: StyleVector;
  playerElo: number;
  exploitationRate: number;     // 0.20 in production
  perturbationStrength: number;   // 0.5 default, tunable
}

export async function getMirrorMove(fen: string, cfg: MirrorConfig): Promise<string> {
  const maiaElo = nearestMaiaElo(cfg.playerElo);
  const motif = detectMotifInPosition(fen, cfg.styleVector);
  
  // Layer 3: Opportunistic Stockfish exploitation
  if (motif && motif.confidence > MOTIF_THRESHOLD && Math.random() < cfg.exploitationRate) {
    const exploitMove = await getExploitMove(fen, motif, cfg.styleVector);
    if (exploitMove) return exploitMove;
  }
  
  // Layer 1 + 2: Maia base + style reranking
  const maiaPolicy = await getMaiaPolicy(fen, maiaElo);
  const rerankedPolicy = perturbPolicy(maiaPolicy, cfg.styleVector, fen, cfg.perturbationStrength);
  
  return sampleFromPolicy(rerankedPolicy);
}

function nearestMaiaElo(playerElo: number): number {
  const bands = [1100, 1300, 1500, 1700, 1900];
  return bands.reduce((nearest, b) => Math.abs(b - playerElo) < Math.abs(nearest - playerElo) ? b : nearest, bands[0]);
}

function sampleFromPolicy(policy: { move: string; prob: number }[]): string {
  const total = policy.reduce((s, p) => s + p.prob, 0);
  let r = Math.random() * total;
  for (const p of policy) {
    r -= p.prob;
    if (r <= 0) return p.move;
  }
  return policy[0].move;
}

async function getExploitMove(
  fen: string,
  motif: { type: string; confidence: number },
  vec: StyleVector
): Promise<string | null> {
  // Only use Stockfish exploitation if the player is historically weak on this motif
  const playerWeakness = vec.motif_blindness[motif.type as keyof typeof vec.motif_blindness];
  if (playerWeakness < 0.3) return null;  // not weak enough — skip
  
  // Get Stockfish's strongest move, which likely creates/exploits the motif
  return await getBestMove(fen, 12);
}

Perturbation function (Layer 2)

src/ml/perturbPolicy.ts 
import { StyleVector } from "./styleVector";
import { Chess } from "chess.js";

export interface MovePolicy {
  move: string;       // UCI notation: "e2e4"
  prob: number;
}

export function perturbPolicy(
  maiaPolicy: MovePolicy[],
  vec: StyleVector,
  fen: string,
  strength: number = 0.5
): MovePolicy[] {
  const game = new Chess(fen);
  const phase = detectPhase(game);   // 'opening' | 'middlegame' | 'endgame'
  
  const perturbed = maiaPolicy.map((entry) => {
    let bonus = 1.0;
    const move = parseMoveSan(game, entry.move);
    if (!move) return entry;
    
    // Opening repertoire bias
    if (phase === "opening") {
      const openingMatch = matchesPlayerRepertoire(game, move, vec);
      if (openingMatch) bonus *= 1 + (0.4 * strength);
    }
    
    // Exchange willingness
    if (isEqualExchange(game, move)) {
      if (vec.exchange_willingness > 0.6) bonus *= 1 + (0.25 * strength);
      else if (vec.exchange_willingness < 0.4) bonus *= 1 - (0.25 * strength);
    }
    
    // Preferred minor piece
    if (move.piece === "n" && vec.preferred_minor === "knight") bonus *= 1 + (0.15 * strength);
    if (move.piece === "b" && vec.preferred_minor === "bishop") bonus *= 1 + (0.15 * strength);
    
    // King safety priority (early castling)
    if (isCastlingMove(move) && vec.king_safety_priority > 0.7 && game.moveNumber() < 12) {
      bonus *= 1 + (0.3 * strength);
    }
    
    return { move: entry.move, prob: entry.prob * bonus };
  });
  
  // Renormalize
  const total = perturbed.reduce((s, p) => s + p.prob, 0);
  return perturbed.map((p) => ({ move: p.move, prob: p.prob / total }));
}

function detectPhase(game: Chess): "opening" | "middlegame" | "endgame" {
  const moveNum = game.moveNumber();
  if (moveNum <= 12) return "opening";
  const material = countMaterial(game);
  if (material <= 14) return "endgame";
  return "middlegame";
}
